cmake_minimum_required(VERSION 3.0.2)
project(shlite_custom_rviz)
find_package(catkin REQUIRED COMPONENTS moveit_ros_visualization moveit_core moveit_msgs pluginlib roscpp rviz)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
catkin_package(LIBRARIES shlite_custom_rviz CATKIN_DEPENDS moveit_ros_visualization moveit_core moveit_msgs pluginlib roscpp rviz)
add_library(shlite_custom_rviz SHARED src/goal_display.cpp)
target_compile_features(shlite_custom_rviz PRIVATE cxx_std_14)
target_include_directories(shlite_custom_rviz PRIVATE ${catkin_INCLUDE_DIRS})
target_link_libraries(shlite_custom_rviz ${catkin_LIBRARIES} Qt5::Widgets)
